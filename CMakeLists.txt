cmake_minimum_required(VERSION 3.21)
project(littlechat VERSION 1.0.0 LANGUAGES CXX)

if(DEFINED CMAKE_CXX_STANDARD AND CMAKE_CXX_STANDARD GREATER 17)
    message(FATAL_ERROR "littlechat supports C++17 at most; do not raise the language standard.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

option(LITTLECHAT_BUILD_CLIENT "Build the Qt desktop client" ON)
option(LITTLECHAT_BUILD_SERVER "Build the Linux diagnostic server" OFF)
option(LITTLECHAT_BUILD_PROTOCOL "Build the shared Protobuf protocol independently" OFF)

if(LITTLECHAT_BUILD_SERVER AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The server targets Ubuntu Linux. Configure the Windows Qt client separately.")
endif()
if(LITTLECHAT_BUILD_PROTOCOL OR LITTLECHAT_BUILD_SERVER)
    add_subdirectory(protocol)
endif()
if(LITTLECHAT_BUILD_SERVER)
    add_subdirectory(server)
endif()
if(LITTLECHAT_BUILD_CLIENT)
    add_subdirectory(client)
endif()
