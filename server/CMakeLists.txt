cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(littlechat_server_project VERSION 1.0.0 LANGUAGES CXX)
    include(CTest)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The server requires Ubuntu Linux")
endif()
if(DEFINED CMAKE_CXX_STANDARD AND CMAKE_CXX_STANDARD GREATER 17)
    message(FATAL_ERROR "littlechat supports C++17 at most")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(LITTLECHAT_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
include("${LITTLECHAT_REPO_ROOT}/cmake/LinuxDependencies.cmake")
if(NOT TARGET littlechat_protocol)
    add_subdirectory("${LITTLECHAT_REPO_ROOT}/protocol" "${CMAKE_BINARY_DIR}/protocol")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/cmake/Muduo.cmake")

add_library(littlechat_concurrency STATIC src/concurrency/boundedthreadpool.cpp src/concurrency/boundedthreadpool.h)
target_include_directories(littlechat_concurrency PUBLIC src)
target_link_libraries(littlechat_concurrency PUBLIC littlechat_build_options Threads::Threads)
target_compile_options(littlechat_concurrency PRIVATE -Wall -Wextra -Wpedantic)

add_executable(littlechat_server src/main.cpp src/network/diagnosticserver.cpp src/network/diagnosticserver.h)
target_include_directories(littlechat_server PRIVATE src)
target_link_libraries(littlechat_server PRIVATE littlechat_protocol littlechat_muduo_net)
target_compile_options(littlechat_server PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(littlechat_server PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS littlechat_server RUNTIME DESTINATION bin)

if(BUILD_TESTING)
    littlechat_add_googletest()
    add_executable(tst_threadpool tests/tst_threadpool.cpp)
    target_link_libraries(tst_threadpool PRIVATE littlechat_concurrency GTest::gtest_main)
    add_test(NAME threadpool COMMAND tst_threadpool)
    set_tests_properties(threadpool PROPERTIES TIMEOUT 15)
    add_executable(tst_server_integration tests/tst_server_integration.cpp)
    target_link_libraries(tst_server_integration PRIVATE littlechat_protocol GTest::gtest Threads::Threads)
    add_test(NAME server_integration COMMAND tst_server_integration $<TARGET_FILE:littlechat_server>)
    set_tests_properties(server_integration PROPERTIES TIMEOUT 60)
endif()
