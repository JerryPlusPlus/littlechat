cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(littlechat_protocol_project VERSION 1.0.0 LANGUAGES CXX)
    include(CTest)
endif()
if(DEFINED CMAKE_CXX_STANDARD AND CMAKE_CXX_STANDARD GREATER 17)
    message(FATAL_ERROR "littlechat supports C++17 at most")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/LinuxDependencies.cmake")

# This is the installed Ubuntu development baseline, not a production dependency recommendation.
set(LITTLECHAT_PROTOBUF_VERSION "3.21.12" CACHE STRING "Exact Protobuf compiler/runtime development version")
find_package(Protobuf ${LITTLECHAT_PROTOBUF_VERSION} EXACT REQUIRED)
execute_process(COMMAND "${Protobuf_PROTOC_EXECUTABLE}" --version
    RESULT_VARIABLE protoc_result OUTPUT_VARIABLE protoc_version OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT protoc_result EQUAL 0 OR NOT protoc_version STREQUAL "libprotoc ${Protobuf_VERSION}")
    message(FATAL_ERROR "protoc/runtime versions must match exactly")
endif()

protobuf_generate_cpp(protocol_sources protocol_headers chat.proto)
add_library(littlechat_protocol STATIC src/frame.cpp src/envelope.cpp ${protocol_sources} ${protocol_headers})
target_include_directories(littlechat_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(littlechat_protocol PUBLIC protobuf::libprotobuf littlechat_build_options)
if(MSVC)
    target_compile_options(littlechat_protocol PUBLIC /utf-8 PRIVATE /W4)
else()
    target_compile_options(littlechat_protocol PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    littlechat_add_googletest()
    add_executable(tst_protocol tests/tst_protocol.cpp)
    target_link_libraries(tst_protocol PRIVATE littlechat_protocol GTest::gtest_main)
    add_test(NAME protocol COMMAND tst_protocol)
endif()
